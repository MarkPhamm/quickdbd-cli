# AdventureWorks, Microsoft's bicycle-company OLTP sample database; one file per SQL schema.
# Converted to QuickDBD syntax from microsoft/sql-server-samples adventure-works/oltp-install-script/instawdb.sql (MIT, see LICENSE).
# Descriptions come from the database's MS_Description properties.

# Cross-reference table mapping vendors with the products they supply.
ProductVendor
-
# Primary key. Foreign key to Product.ProductID.
ProductID PK int FK >- Product.ProductID
# Primary key. Foreign key to Vendor.BusinessEntityID.
BusinessEntityID PK int FK >- Vendor.BusinessEntityID
# The average span of time (in days) between placing an order with the vendor and receiving the purchased product.
AverageLeadTime int
# The vendor's usual selling price.
StandardPrice money
# The selling price when last purchased.
LastReceiptCost money NULL
# Date the product was last received by the vendor.
LastReceiptDate datetime NULL
# The maximum quantity that should be ordered.
MinOrderQty int
# The minimum quantity that should be ordered.
MaxOrderQty int
# The quantity currently on order.
OnOrderQty int NULL
# The product's unit of measure.
UnitMeasureCode nchar(3) FK >- UnitMeasure.UnitMeasureCode
# Date and time the record was last updated.
ModifiedDate datetime

# Individual products associated with a specific purchase order. See PurchaseOrderHeader.
PurchaseOrderDetail
-
# Primary key. Foreign key to PurchaseOrderHeader.PurchaseOrderID.
PurchaseOrderID PK int FK >- PurchaseOrderHeader.PurchaseOrderID
# Primary key. One line number per purchased product.
PurchaseOrderDetailID PK int IDENTITY
# Date the product is expected to be received.
DueDate datetime
# Quantity ordered.
OrderQty smallint
# Product identification number. Foreign key to Product.ProductID.
ProductID int FK >- Product.ProductID
# Vendor's selling price of a single product.
UnitPrice money
# Per product subtotal. Computed as OrderQty * UnitPrice.
LineTotal computed NULL
# Quantity actually received from the vendor.
ReceivedQty decimal(8,2)
# Quantity rejected during inspection.
RejectedQty decimal(8,2)
# Quantity accepted into inventory. Computed as ReceivedQty - RejectedQty.
StockedQty computed NULL
# Date and time the record was last updated.
ModifiedDate datetime

# General purchase order information. See PurchaseOrderDetail.
PurchaseOrderHeader
-
# Primary key.
PurchaseOrderID PK int IDENTITY
# Incremental number to track changes to the purchase order over time.
RevisionNumber tinyint default=0
# Order current status. 1 = Pending; 2 = Approved; 3 = Rejected; 4 = Complete
Status tinyint default=1
# Employee who created the purchase order. Foreign key to Employee.BusinessEntityID.
EmployeeID int FK >- Employee.BusinessEntityID
# Vendor with whom the purchase order is placed. Foreign key to Vendor.BusinessEntityID.
VendorID int FK >- Vendor.BusinessEntityID
# Shipping method. Foreign key to ShipMethod.ShipMethodID.
ShipMethodID int FK >- ShipMethod.ShipMethodID
# Purchase order creation date.
OrderDate datetime
# Estimated shipment date from the vendor.
ShipDate datetime NULL
# Purchase order subtotal. Computed as SUM(PurchaseOrderDetail.LineTotal)for the appropriate PurchaseOrderID.
SubTotal money default=0.00
# Tax amount.
TaxAmt money default=0.00
# Shipping cost.
Freight money default=0.00
# Total due to vendor. Computed as Subtotal + TaxAmt + Freight.
TotalDue computed NULL
# Date and time the record was last updated.
ModifiedDate datetime

# Shipping company lookup table.
ShipMethod
-
# Primary key for ShipMethod records.
ShipMethodID PK int IDENTITY
# Shipping company name.
Name Name UNIQUE
# Minimum shipping charge.
ShipBase money default=0.00
# Shipping charge per pound.
ShipRate money default=0.00
# ROWGUIDCOL number uniquely identifying the record. Used to support a merge replication sample.
rowguid uniqueidentifier UNIQUE
# Date and time the record was last updated.
ModifiedDate datetime

# Companies from whom Adventure Works Cycles purchases parts or other goods.
Vendor
-
# Primary key for Vendor records.  Foreign key to BusinessEntity.BusinessEntityID
BusinessEntityID PK int FK - BusinessEntity.BusinessEntityID
# Vendor account (identification) number.
AccountNumber AccountNumber UNIQUE
# Company name.
Name Name
# 1 = Superior, 2 = Excellent, 3 = Above average, 4 = Average, 5 = Below average
CreditRating tinyint
# 0 = Do not use if another vendor is available. 1 = Preferred over other vendors supplying the same product.
PreferredVendorStatus Flag default=1
# 0 = Vendor no longer used. 1 = Vendor is actively used.
ActiveFlag Flag default=1
PurchasingWebServiceURL nvarchar(1024) NULL
ModifiedDate datetime
