# AdventureWorks, Microsoft's bicycle-company OLTP sample database; one file per SQL schema.
# Converted to QuickDBD syntax from microsoft/sql-server-samples adventure-works/oltp-install-script/instawdb.sql (MIT, see LICENSE).
# Descriptions come from the database's MS_Description properties.

# Lookup table containing the departments within the Adventure Works Cycles company.
Department
-
# Primary key for Department records.
DepartmentID PK smallint IDENTITY
# Name of the department.
Name Name UNIQUE
# Name of the group to which the department belongs.
GroupName Name
# Date and time the record was last updated.
ModifiedDate datetime

# Employee information such as salary, department, and title.
Employee
-
# Primary key for Employee records.  Foreign key to BusinessEntity.BusinessEntityID.
BusinessEntityID PK int FK - Person.BusinessEntityID
# Unique national identification number such as a social security number.
NationalIDNumber nvarchar(15) UNIQUE
# Network login.
LoginID nvarchar(256) UNIQUE
# Where the employee is located in corporate hierarchy.
OrganizationNode hierarchyid NULL
# The depth of the employee in the corporate hierarchy.
OrganizationLevel computed NULL
# Work title such as Buyer or Sales Representative.
JobTitle nvarchar(50)
# Date of birth.
BirthDate date
# M = Married, S = Single
MaritalStatus nchar(1)
# M = Male, F = Female
Gender nchar(1)
# Employee hired on this date.
HireDate date
# Job classification. 0 = Hourly, not exempt from collective bargaining. 1 = Salaried, exempt from collective bargaining.
SalariedFlag Flag default=1
# Number of available vacation hours.
VacationHours smallint default=0
# Number of available sick leave hours.
SickLeaveHours smallint default=0
# 0 = Inactive, 1 = Active
CurrentFlag Flag default=1
# ROWGUIDCOL number uniquely identifying the record. Used to support a merge replication sample.
rowguid uniqueidentifier UNIQUE
# Date and time the record was last updated.
ModifiedDate datetime

# Employee department transfers.
EmployeeDepartmentHistory
-
# Employee identification number. Foreign key to Employee.BusinessEntityID.
BusinessEntityID PK int FK >- Employee.BusinessEntityID
# Department in which the employee worked including currently. Foreign key to Department.DepartmentID.
DepartmentID PK smallint FK >- Department.DepartmentID
# Identifies which 8-hour shift the employee works. Foreign key to Shift.Shift.ID.
ShiftID PK tinyint FK >- Shift.ShiftID
# Date the employee started work in the department.
StartDate PK date
# Date the employee left the department. NULL = Current department.
EndDate date NULL
# Date and time the record was last updated.
ModifiedDate datetime

# Employee pay history.
EmployeePayHistory
-
# Employee identification number. Foreign key to Employee.BusinessEntityID.
BusinessEntityID PK int FK >- Employee.BusinessEntityID
# Date the change in pay is effective
RateChangeDate PK datetime
# Salary hourly rate.
Rate money
# 1 = Salary received monthly, 2 = Salary received biweekly
PayFrequency tinyint
# Date and time the record was last updated.
ModifiedDate datetime

# Résumés submitted to Human Resources by job applicants.
JobCandidate
-
# Primary key for JobCandidate records.
JobCandidateID PK int IDENTITY
# Employee identification number if applicant was hired. Foreign key to Employee.BusinessEntityID.
BusinessEntityID int NULL FK >- Employee.BusinessEntityID
# Résumé in XML format.
Resume XML(HumanResources.HRResumeSchemaCollection) NULL
# Date and time the record was last updated.
ModifiedDate datetime

# Work shift lookup table.
Shift
-
# Primary key for Shift records.
ShiftID PK tinyint IDENTITY
# Shift description.
Name Name UNIQUE
# Shift start time.
StartTime time
# Shift end time.
EndTime time
# Date and time the record was last updated.
ModifiedDate datetime
