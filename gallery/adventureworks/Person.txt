# Street address information for customers, employees, and vendors.
Address
-
# Primary key for Address records.
AddressID PK int IDENTITY
# First street address line.
AddressLine1 nvarchar(60)
# Second street address line.
AddressLine2 nvarchar(60) NULL
# Name of the city.
City nvarchar(30)
# Unique identification number for the state or province. Foreign key to StateProvince table.
StateProvinceID int FK >- StateProvince.StateProvinceID
# Postal code for the street address.
PostalCode nvarchar(15)
# Latitude and longitude of this address.
SpatialLocation geography NULL
# ROWGUIDCOL number uniquely identifying the record. Used to support a merge replication sample.
rowguid uniqueidentifier UNIQUE
# Date and time the record was last updated.
ModifiedDate datetime

# Human beings involved with AdventureWorks: employees, customer contacts, and vendor contacts.
Person
-
# Primary key for Person records.
BusinessEntityID PK int FK - BusinessEntity.BusinessEntityID
# Primary type of person: SC = Store Contact, IN = Individual (retail) customer, SP = Sales person, EM = Employee (non-sales), VC = Vendor contact, GC = General contact
PersonType nchar(2)
# 0 = The data in FirstName and LastName are stored in western style (first name, last name) order.  1 = Eastern style (last name, first name) order.
NameStyle NameStyle default=0
# A courtesy title. For example, Mr. or Ms.
Title nvarchar(8) NULL
# First name of the person.
FirstName Name
# Middle name or middle initial of the person.
MiddleName Name NULL
# Last name of the person.
LastName Name
# Surname suffix. For example, Sr. or Jr.
Suffix nvarchar(10) NULL
# 0 = Contact does not wish to receive e-mail promotions, 1 = Contact does wish to receive e-mail promotions from AdventureWorks, 2 = Contact does wish to receive e-mail promotions from AdventureWorks and selected partners.
EmailPromotion int default=0
# Additional contact information about the person stored in xml format.
AdditionalContactInfo XML(Person.AdditionalContactInfoSchemaCollection) NULL
# Personal information such as hobbies, and income collected from online shoppers. Used for sales analysis.
Demographics XML(Person.IndividualSurveySchemaCollection) NULL
# ROWGUIDCOL number uniquely identifying the record. Used to support a merge replication sample.
rowguid uniqueidentifier UNIQUE
# Date and time the record was last updated.
ModifiedDate datetime
